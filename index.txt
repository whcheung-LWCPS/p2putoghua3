        function initAudio() {
            if (!audioCtx) {

        // PCM 轉 WAV 格式輔助函式 (給 Gemini TTS 使用)
        function base64ToArrayBuffer(base64) {
            const binaryString = window.atob(base64);
            const len = binaryString.length;
            const bytes = new Uint8Array(len);
            for (let i = 0; i < len; i++) {
                bytes[i] = binaryString.charCodeAt(i);
            }
            return bytes.buffer;
        }

        function pcmToWav(pcm16, sampleRate = 24000) {
            const buffer = new ArrayBuffer(44 + pcm16.length * 2);
            const view = new DataView(buffer);

            function writeString(offset, string) {
                for (let i = 0; i < string.length; i++) {
                    view.setUint8(offset + i, string.charCodeAt(i));
                }
            }

            writeString(0, 'RIFF');
            view.setUint32(4, 36 + pcm16.length * 2, true);
            writeString(8, 'WAVE');
            writeString(12, 'fmt ');
            view.setUint32(16, 16, true);
            view.setUint16(20, 1, true); // PCM
            view.setUint16(22, 1, true); // 單聲道
            view.setUint32(24, sampleRate, true);
            view.setUint32(28, sampleRate * 2, true);
            view.setUint16(32, 2, true);
            view.setUint16(34, 16, true);
            writeString(36, 'data');
            view.setUint32(40, pcm16.length * 2, true);

            let offset = 44;
            for (let i = 0; i < pcm16.length; i++, offset += 2) {
                view.setInt16(offset, pcm16[i], true);
            }

            return new Blob([buffer], { type: 'audio/wav' });
        }

        async function fetchGeminiMandarinTTS(text) {
            const apiKey = "";
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-tts:generateContent?key=${apiKey}`;
            
            const payload = {
                contents: [{
                    parts: [{ text: `請用標準、清晰的中國普通話朗讀這個詞語：${text}` }]
                }],
                generationConfig: {
                    responseModalities: ["AUDIO"],
                    speechConfig: {
                        voiceConfig: {
                            prebuiltVoiceConfig: { voiceName: "Leda" }
                        }
                    }
                }
            };

            const response = await fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });

            const result = await response.json();
            const candidate = result.candidates?.[0];
            const part = candidate?.content?.parts?.[0];
            const audioData = part?.inlineData?.data;
            const mimeType = part?.inlineData?.mimeType;

            if (audioData && mimeType) {
                const sampleRateMatch = mimeType.match(/rate=(\d+)/);
                const sampleRate = sampleRateMatch ? parseInt(sampleRateMatch[1], 10) : 24000;
                const pcmBytes = base64ToArrayBuffer(audioData);
                const pcm16 = new Int16Array(pcmBytes);
                const wavBlob = pcmToWav(pcm16, sampleRate);
                return URL.createObjectURL(wavBlob);
            }
            throw new Error('Gemini TTS audio error');
        }

        async function speakMandarin(text, event) {
            if (event) event.stopPropagation();
            initAudio();

            // 停止當前正在播放的聲音
            if (currentMandarinAudio) {
                currentMandarinAudio.pause();
                currentMandarinAudio = null;
            }
            if ('speechSynthesis' in window) {
                window.speechSynthesis.cancel();
            }

            // 1. 檢查快取（已讀取的語音秒播）
            if (audioCache[text]) {
                playAudioUrl(audioCache[text]);
                return;
            }

            // 2. 優先調用 Gemini 2.5 Flash TTS (100% 標準普通話)
            try {
                const wavUrl = await fetchGeminiMandarinTTS(text);
                audioCache[text] = wavUrl;
                playAudioUrl(wavUrl);
                return;
            } catch (err) {
                console.warn('Gemini TTS 網路無法調用，切換至備用普通話語音庫:', err);
            }

            // 3. 備用線上標準普通話 API
            const encodedText = encodeURIComponent(text);
            const primaryTtsUrl = `https://dict.youdao.com/dictvoice?le=zh&audio=${encodedText}`;
            const secondaryTtsUrl = `https://translate.google.com/translate_tts?ie=UTF-8&q=${encodedText}&tl=zh-CN&client=tw-ob`;

            const audio = new Audio(primaryTtsUrl);
            currentMandarinAudio = audio;
            audio.playbackRate = 0.85;

            audio.play().then(() => {
                audioCache[text] = primaryTtsUrl;
            }).catch(() => {
                const backupAudio = new Audio(secondaryTtsUrl);
                currentMandarinAudio = backupAudio;
                backupAudio.playbackRate = 0.85;
                backupAudio.play().then(() => {
                    audioCache[text] = secondaryTtsUrl;
                }).catch(() => {
                    speakMandarinOfflineStrict(text);
                });
            });
        }

        function playAudioUrl(url) {
            const audio = new Audio(url);
            currentMandarinAudio = audio;
            audio.playbackRate = 0.85;
            audio.play().catch(() => {});
        }

        function speakMandarinOfflineStrict(text) {
            if (!('speechSynthesis' in window)) return;

            const voices = window.speechSynthesis.getVoices();
            // 嚴格篩選普通話發音，堅決排除所有廣東話/粵語/zh-HK/Cantonese
            const mandarinVoice = voices.find(v => 
                (v.lang === 'zh-CN' || v.lang === 'zh_CN' || v.lang.startsWith('zh-Hans') || v.lang.includes('Mandarin') || v.lang.includes('Putonghua') || v.name.includes('Mainland') || v.name.includes('Huihui') || v.name.includes('Ting-Ting') || v.name.includes('Yaoyao')) &&
                !v.lang.includes('HK') && !v.name.toLowerCase().includes('cantonese') && !v.name.toLowerCase().includes('hong kong')
            );

            if (!mandarinVoice) {
                console.warn('未能在本機找到明確的普通話語音庫，已終止離線播放以防止誤播廣東話。');
                return;
            }

            const utterance = new SpeechSynthesisUtterance(text);
            utterance.rate = 0.8;
            utterance.voice = mandarinVoice;
            utterance.lang = 'zh-CN';

            window.speechSynthesis.speak(utterance);
        }

        if ('speechSynthesis' in window) {
            window.speechSynthesis.onvoiceschanged = () => {
                window.speechSynthesis.getVoices();
            };
        }

        function playSound(type) {
            if (!state.soundEnabled) return;
            initAudio();
            if (!audioCtx) return;

            try {
                if (type === 'correct') {
                    const now = audioCtx.currentTime;
                    const osc = audioCtx.createOscillator();
                    const gain = audioCtx.createGain();
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(523.25, now); // C5
                    osc.frequency.setValueAtTime(659.25, now + 0.12); // E5
                    gain.gain.setValueAtTime(0.2, now);
                    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.35);
                    osc.connect(gain);
                    gain.connect(audioCtx.destination);
                    osc.start(now);
                    osc.stop(now + 0.35);
                } else if (type === 'wrong') {
                    const now = audioCtx.currentTime;
                    const osc = audioCtx.createOscillator();
                    const gain = audioCtx.createGain();
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(180, now);
                    osc.frequency.linearRampToValueAtTime(120, now + 0.25);
                    gain.gain.setValueAtTime(0.25, now);
                    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.25);
                    osc.connect(gain);
                    gain.connect(audioCtx.destination);
                    osc.start(now);
                    osc.stop(now + 0.25);
                } else if (type === 'win') {
                    const notes = [523.25, 659.25, 783.99, 1046.50];
                    notes.forEach((freq, i) => {
                        const now = audioCtx.currentTime + i * 0.12;
                        const osc = audioCtx.createOscillator();
                        const gain = audioCtx.createGain();
                        osc.type = 'triangle';
                        osc.frequency.setValueAtTime(freq, now);
                        gain.gain.setValueAtTime(0.3, now);
                        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.3);
                        osc.connect(gain);
                        gain.connect(audioCtx.destination);
                        osc.start(now);
                        osc.stop(now + 0.3);
                    });
                }
            } catch (e) {
                console.warn('Audio playback error:', e);
            }
        }

        function handleDragStart(e, id) {
            initAudio();
            state.draggedItemId = id;
            e.dataTransfer.setData('text/plain', id);
            e.dataTransfer.effectAllowed = 'move';
        }

        function handleDragOver(e) {
            e.preventDefault();
            e.dataTransfer.dropEffect = 'move';
        }

        function handleDragLeave(e) {
            e.preventDefault();
        }

        function handleDrop(e, category) {
            e.preventDefault();
            const itemId = e.dataTransfer.getData('text/plain') || state.draggedItemId;
            if (itemId) {
                processClassification(itemId, category);
                state.draggedItemId = null;
            }
        }

        function initGame() {
            // 洗牌隨機打亂物品順序
            const shuffled = [...ALL_ITEMS].sort(() => Math.random() - 0.5);

            state.remainingItems = shuffled;
            state.hotItems = [];
            state.rainyItems = [];
            state.selectedItemId = null;
            state.draggedItemId = null;
            state.score = 0;

            updateScoreUI();
            renderAllPools();
            showHint('💡 點擊 🔊 聽普通話，把物品拖曳或點擊放進「天氣炎熱」或「下雨」欄位！');
        }

        function restartGame() {
            initGame();
        }

        function toggleSound() {
            state.soundEnabled = !state.soundEnabled;
            const btn = document.getElementById('sound-toggle');
            btn.textContent = state.soundEnabled ? '🔊' : '🔇';
            btn.title = state.soundEnabled ? '已開啟音效' : '已靜音';
        }

        function showHint(text) {
            const hintBox = document.getElementById('hint-box');
            if (hintBox) {
                hintBox.textContent = text;
            }
        }

        function triggerScoreEffect(score) {
            const display = document.getElementById('score-display');
            if (!display) return;
            const floatEl = document.createElement('span');
            floatEl.className = 'absolute -top-4 right-0 text-orange-600 font-black text-xs float-score pointer-events-none';
            floatEl.textContent = `+${score}`;
            display.parentElement.appendChild(floatEl);
            setTimeout(() => floatEl.remove(), 700);
        }

        function updateScoreUI() {
            document.getElementById('score-display').textContent = state.score;
        }

        function renderAllPools() {
            const isCompleted = state.remainingItems.length === 0;
            const mainContainer = document.getElementById('main-container');
            const poolColumn = document.getElementById('pool-column');

            // 完成挑戰時：隱藏待分類欄，將主佈局改為放大雙欄 (grid-cols-2)
            if (isCompleted) {
                if (poolColumn) poolColumn.classList.add('hidden');
                if (mainContainer) {
                    mainContainer.classList.remove('grid-cols-3');
                    mainContainer.classList.add('grid-cols-2');
                }
            } else {
                if (poolColumn) poolColumn.classList.remove('hidden');
                if (mainContainer) {
                    mainContainer.classList.remove('grid-cols-2');
                    mainContainer.classList.add('grid-cols-3');
                }
            }

            // 1. 渲染右側待分類區
            const poolEl = document.getElementById('item-pool');
            poolEl.innerHTML = '';

            if (!isCompleted) {
                state.remainingItems.forEach(item => {
                    const isSelected = state.selectedItemId === item.id;
                    const card = document.createElement('div');
                    card.id = `item-${item.id}`;
                    card.draggable = true;
                    card.className = `bg-white rounded-lg p-1 border-2 cursor-pointer transition-all duration-150 flex flex-col items-center justify-between gap-0.5 shadow-2xs hover:shadow-xs ${
                        isSelected 
                        ? 'border-orange-500 bg-orange-50 ring-1 ring-orange-200 scale-[1.01]' 
                        : 'border-amber-100 hover:border-amber-300'
                    }`;

                    card.ondragstart = (e) => handleDragStart(e, item.id);
                    card.onclick = (e) => {
                        e.stopPropagation();
                        handleItemClick(item.id);
                    };

                    card.innerHTML = `
                        <div class="pointer-events-none shrink-0 transform scale-75 sm:scale-90 origin-center">${item.iconSvg}</div>
                        <div class="text-center pointer-events-none w-full min-w-0">
                            <span class="text-[9px] sm:text-[10px] text-amber-700 font-bold block leading-tight truncate">${item.pinyin}</span>
                            <span class="font-black text-xs sm:text-sm text-slate-800 block leading-tight truncate">${item.name}</span>
                        </div>
                        <button onclick="speakMandarin('${item.name}', event)" class="p-0.5 px-2 bg-amber-100 hover:bg-amber-200 active:scale-95 text-amber-900 rounded-full text-[10px] font-bold transition flex items-center justify-center" title="收聽普通話發音">
                            🔊
                        </button>
                    `;
                    poolEl.appendChild(card);
                });
            }

            // 2. 渲染左側天氣炎熱區
            const hotEl = document.getElementById('hot-items');
            hotEl.innerHTML = '';
            state.hotItems.forEach(item => {
                hotEl.appendChild(createPlacedCard(item, 'hot', isCompleted));
            });

            // 3. 渲染中間下雨區
            const rainyEl = document.getElementById('rainy-items');
            rainyEl.innerHTML = '';
            state.rainyItems.forEach(item => {
                rainyEl.appendChild(createPlacedCard(item, 'rainy', isCompleted));
            });

            // 4. 更新數量指標
            document.getElementById('remaining-count').textContent = state.remainingItems.length;
            document.getElementById('hot-count').textContent = `${state.hotItems.length} 件`;
            document.getElementById('rainy-count').textContent = `${state.rainyItems.length} 件`;

            // 5. 更新進度條
            const total = ALL_ITEMS.length;
            const completed = total - state.remainingItems.length;
            const percent = Math.round((completed / total) * 100);
            document.getElementById('progress-bar').style.width = `${percent}%`;
            document.getElementById('progress-text').textContent = `${completed} / ${total}`;
        }

        function createPlacedCard(item, zone, isCompleted = false) {
            const card = document.createElement('div');
            const isHot = zone === 'hot';

            if (isCompleted) {
                // 完成後的放大版卡片
                card.className = `animate-pop rounded-xl p-2 flex items-center gap-2 border shadow-2xs ${
                    isHot ? 'bg-amber-100/95 border-amber-300 text-orange-950' : 'bg-blue-100/95 border-blue-300 text-blue-950'
                }`;
                card.innerHTML = `
                    <div class="w-8 h-8 shrink-0 flex items-center justify-center transform scale-90">${item.iconSvg}</div>
                    <div class="flex flex-col min-w-0 flex-1">
                        <span class="text-xs ${isHot ? 'text-orange-800' : 'text-blue-800'} font-black leading-tight truncate">${item.pinyin}</span>
                        <span class="font-black text-base sm:text-lg text-slate-900 truncate leading-tight">${item.name}</span>
                    </div>
                    <button onclick="speakMandarin('${item.name}', event)" class="p-1 px-2.5 bg-white hover:bg-slate-50 text-slate-800 rounded-lg text-xs font-black shadow-2xs transition active:scale-95 flex items-center justify-center shrink-0 border border-slate-200" title="收聽普通話發音">
                        🔊
                    </button>
                `;
            } else {
                // 一般遊玩中卡片
                card.className = `animate-pop rounded-lg p-1 flex items-center gap-1.5 border shadow-2xs ${
                    isHot ? 'bg-amber-100/90 border-amber-200 text-orange-950' : 'bg-blue-100/90 border-blue-200 text-blue-950'
                }`;
                card.innerHTML = `
                    <div class="w-6 h-6 shrink-0 flex items-center justify-center transform scale-75">${item.iconSvg}</div>
                    <div class="flex flex-col min-w-0 flex-1">
                        <span class="text-[9px] ${isHot ? 'text-orange-700' : 'text-blue-700'} font-bold leading-none truncate">${item.pinyin}</span>
                        <span class="font-black text-xs sm:text-sm truncate leading-tight">${item.name}</span>
                    </div>
                    <button onclick="speakMandarin('${item.name}', event)" class="p-0.5 px-1.5 bg-white/80 hover:bg-white text-slate-700 rounded-md text-[10px] shadow-2xs transition active:scale-90 flex items-center justify-center shrink-0" title="收聽普通話發音">
                        🔊
                    </button>
                `;
            }
            return card;
        }

        function handleItemClick(id) {
            initAudio();
            if (state.selectedItemId === id) {
                state.selectedItemId = null;
                showHint('💡 已取消選取');
            } else {
                state.selectedItemId = id;
                const item = ALL_ITEMS.find(i => i.id === id);
                showHint(`👉 已選取「${item.name}」，請點擊「天氣炎熱」或「下雨」欄位放置！`);
            }
            renderAllPools();
        }

        function handleZoneClick(targetCategory) {
            if (state.selectedItemId) {
                processClassification(state.selectedItemId, targetCategory);
                state.selectedItemId = null;
            }
        }

        function processClassification(itemId, targetCategory) {
            const item = ALL_ITEMS.find(i => i.id === itemId);
            if (!item) return;

            if (item.category === targetCategory) {
                playSound('correct');
                
                const addedScore = 10;
                state.score += addedScore;
                
                state.remainingItems = state.remainingItems.filter(i => i.id !== itemId);
                if (targetCategory === 'hot') {
                    state.hotItems.push(item);
                } else {
                    state.rainyItems.push(item);
                }

                const categoryText = targetCategory === 'hot' ? '天氣炎熱' : '下雨';
                showHint(`✨ 太棒了！「${item.name}」適合在【${categoryText}】時使用！${item.hint}`);
                triggerScoreEffect(addedScore);
                updateScoreUI();
                renderAllPools();

                // 檢查是否全部分類完成
                if (state.remainingItems.length === 0) {
                    setTimeout(() => {
                        playSound('win');
                        showHint('🎉 恭喜你完成了所有物品配對！隨時點擊 🔊 重複聽普通話發音喔！');
                    }, 400);
                }

            } else {
                playSound('wrong');

                const cardEl = document.getElementById(`item-${itemId}`);
                if (cardEl) {
                    cardEl.classList.add('animate-shake', 'border-red-500');
                    setTimeout(() => {
                        cardEl.classList.remove('animate-shake', 'border-red-500');
                    }, 400);
                }

                const correctName = item.category === 'hot' ? '天氣炎熱' : '下雨';
                showHint(`❌ 「${item.name}」其實是【${correctName}】時使用的物品喔！再試一次吧！`);
            }
        }

        // 頁面載入後自動初始化遊戲
        window.addEventListener('DOMContentLoaded', initGame);
        window.onload = initGame;
    </script>
